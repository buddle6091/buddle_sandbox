#### 문제
```javascript
function solution(num) {
    var answer = '';
    return answer;
}
```
---
```
💡 num 을 2로 나눴을 때 나머지가 0 이면, 짝수 아니면 홀수
```
---
#### ✨ 풀이 방법
```javascript
function solution(num) {
    var answer = '';
  // 조건문을 써서 요건에 따른 리턴값을 다르게 지정할 수 있다. 여기서 if 대신 삼항 연산자를 사용
 // num 을 2로 나눈 나머지가 0일때 와 아닐 때의 리턴 값을 다르게 지정
    answer = (num % 2 == 0) ? 'Even' : 'Odd'
    return answer;
}
```
